    47→/* =============================================================== database */
    48→function db(): PDO
    49→{
    50→    static $pdo = null;
    51→    if ($pdo === null) {
    52→        if (!is_file(MUHDIN_DB_PATH)) {
    53→            fail('Database tidak ditemukan di ' . basename(dirname(MUHDIN_DB_PATH)) . '/ — pastikan data/muhdin.sqlite ikut terunggah.', 500);
    54→        }
    55→        $pdo = new PDO('sqlite:' . MUHDIN_DB_PATH, null, null, [
    56→            PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    57→            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    58→        ]);
    59→        try { $pdo->exec('PRAGMA journal_mode=WAL'); } catch (Throwable $e) { /* NFS: biarkan mode bawaan */ }
    60→        $pdo->exec('PRAGMA busy_timeout=5000');
    61→        $pdo->exec('PRAGMA foreign_keys=ON');
    62→        // Tabel pendukung backend PHP (rate limit).
    63→        $pdo->exec('CREATE TABLE IF NOT EXISTS RateLimit (key TEXT PRIMARY KEY, hits TEXT NOT NULL DEFAULT "[]")');
    64→    }
    65→    return $pdo;
    66→}

   111→/** epoch-ms → ISO-8601 UTC dengan presisi milidetik (gaya serialisasi Prisma). */
   112→function iso_date(?int $ms): ?string
   113→{
   114→    if ($ms === null) return null;
   115→    $sec = (int) floor($ms / 1000);
   116→    return gmdate('Y-m-d\TH:i:s', $sec) . '.' . sprintf('%03d', $ms % 1000) . 'Z';
   117→}

   119→/** Nilai tanggal masuk (string ISO / angka epoch) → epoch-ms untuk kolom SQLite. */
   120→function date_to_ms($value): ?int
   121→{
   122→    if ($value === null || $value === '') return null;
   123→    if (is_numeric($value)) return (int) $value;
   124→    $t = strtotime((string) $value);
   125→    return $t === false ? null : $t * 1000;

   196→/** Normalisasi satu baris SQLite → bentuk JSON ala Prisma (ISO date, bool, int). */
   197→function cast_row(string $table, array $row): array
   198→{
   199→    foreach (DATE_COLS[$table] ?? [] as $c) {
   200→        if (array_key_exists($c, $row) && $row[$c] !== null) {
   201→            $row[$c] = is_numeric($row[$c]) ? iso_date((int) $row[$c]) : $row[$c];
   202→        }
   203→    }
   204→    foreach (BOOL_COLS[$table] ?? [] as $c) {
   205→        if (array_key_exists($c, $row) && $row[$c] !== null) {
   206→            $row[$c] = (bool) ((int) $row[$c]);
   207→        }
   208→    }
   209→    foreach (INT_COLS[$table] ?? [] as $c) {
   210→        if (array_key_exists($c, $row) && $row[$c] !== null && is_numeric($row[$c])) {
   211→            $row[$c] = (int) $row[$c];
   212→        }
   213→    }
   214→    return $row;
   215→}
   216→
   217→function cast_rows(string $table, array $rows): array
   218→{
   219→    return array_map(static fn (array $r): array => cast_row($table, $r), $rows);

   271→/* ================================================================== router */
   272→final class Router
   273→{
   274→    /** @var array<string, array<string, callable>> */
   275→    private array $routes = [];
   276→
   277→    public function on(string $method, string $pattern, callable $handler): void
   278→    {
   279→        $this->routes[strtoupper($method)][$pattern] = $handler;
   280→    }
   281→
   282→    /** Pattern: segmen literal, ':param', dan '*' (sisa segmen). */
   283→    public function dispatch(string $method, string $path): void
   284→    {
   285→        $segs = $path === '' ? [] : explode('/', trim($path, '/'));
   286→        foreach (['ANY', strtoupper($method)] as $m) {
   287→            foreach ($this->routes[$m] ?? [] as $pattern => $handler) {
   288→                $params = $this->match($pattern === '' ? [] : explode('/', $pattern), $segs);
   289→                if ($params !== null) {
   290→                    $handler($params);
   291→                    return;
   292→                }
   293→            }
   294→        }
   295→        fail('Endpoint tidak ditemukan: ' . strtoupper($method) . ' /' . implode('/', $segs), 404);
   296→    }
   297→
   298→    /** @return array<string,string>|null */
   299→    private function match(array $pattern, array $segs): ?array
   300→    {
   301→        $params = [];
   302→        $i = 0;
   303→        foreach ($pattern as $ps) {
   304→            if ($ps === '*') {
   305→                $rest = array_slice($segs, $i);
   306→                if ($rest === []) return null;
   307→                $params['rest'] = implode('/', $rest);
   308→                return $params;
   309→            }
   310→            if (!array_key_exists($i, $segs)) return null;
   311→            if ($ps !== '' && $ps[0] === ':') {
   312→                $params[substr($ps, 1)] = rawurldecode($segs[$i]);
   313→            } elseif ($ps !== $segs[$i]) {
   314→                return null;
   315→            }
   316→            $i++;
   317→        }
   318→        return count($segs) === count($pattern) ? $params : null;
   319→    }
   320→}

   323→/**
   324→ * Mirip getSessionUser(): cookie muhdin_session → baris Session+User.
   325→ * Sesi kedaluwarsa / user nonaktif dibersihkan (paritas Task 15-c).
   326→ * Return baris User penuh (termasuk kolom password — handler yang memilah).
   327→ */
   328→function current_user(): ?array
   329→{
   330→    try {
   331→        $token = read_cookie(MUHDIN_SESSION_COOKIE);
   332→        if ($token === null) return null;
   333→        $row = q_one(
   334→            'SELECT u.*, s.id AS __sessionId, s.expiresAt AS __expiresAt FROM Session s JOIN User u ON u.id = s.userId WHERE s.token = ? LIMIT 1',
   335→            [$token]
   336→        );
   337→        if ($row === null) return null;
   338→        if ((int) $row['__expiresAt'] < now_ms()) {
   339→            q_exec('DELETE FROM Session WHERE id = ?', [$row['__sessionId']]);
   340→            return null;
   341→        }
   342→        if (!((int) $row['isActive'])) {
   343→            q_exec('DELETE FROM Session WHERE userId = ?', [$row['id']]);
   344→            return null;
   345→        }
   346→        return $row;
   347→    } catch (Throwable $e) {
   348→        return null;
   349→    }
   350→}

   352→/**
   353→ * Verifikasi password. Hash aktif di edisi PHP = bcrypt ($2y$, password_hash).
   354→ * Hash scrypt warisan versi Node ("salt:hash") tidak dapat diverifikasi PHP —
   355→ * akun demo di-rehash bcrypt saat build, akun lain direset lewat CMS.
   356→ */
   357→function verify_password(string $password, string $stored): bool
   358→{
   359→    if ($password === '' || $stored === '') return false;
   360→    if (strpos($stored, ':') !== false && strpos($stored, '$') !== 0) {
   361→        return false; // format scrypt Node → tidak didukung PHP
   362→    }
   363→    $hash = str_starts_with($stored, '$2b$') || str_starts_with($stored, '$2a$')
   364→        ? '$2y$' . substr($stored, 4)
   365→        : $stored;
   366→    return password_verify($password, $hash);
   367→}

   374→/** Buat sesi baru + set cookie (paritas createSession Node). */
   375→function create_session(string $userId): string
   376→{
   377→    $token = bin2hex(random_bytes(32));
   378→    $now = now_ms();
   379→    q_exec(
   380→        'INSERT INTO Session (id, token, userId, expiresAt, createdAt) VALUES (?,?,?,?,?)',
   381→        [new_id(), $token, $userId, $now + MUHDIN_SESSION_DAYS * 86400000, $now]
   382→    );
   383→    $https = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off')
   384→        || (int) ($_SERVER['SERVER_PORT'] ?? 0) === 443;
   385→    setcookie(MUHDIN_SESSION_COOKIE, $token, [
   386→        'expires'  => time() + MUHDIN_SESSION_DAYS * 86400,
   387→        'path'     => '/',
   388→        'secure'   => $https,
   389→        'httponly' => true,
   390→        'samesite' => 'Lax',
   391→    ]);
   392→    return $token;
   393→}

   471→/* ============================================================== rate limit */
   472→/** Paritas rateLimit Node: bucket:ip, default 5 hit / 60 detik. */
   473→function rate_limit(string $bucket, int $max = 5, int $windowMs = 60000): bool
   474→{
   475→    try {
   476→        $key = $bucket . ':' . client_ip();
   477→        $now = now_ms();
   478→        $row = q_one('SELECT hits FROM RateLimit WHERE key = ?', [$key]);
   479→        $hits = [];
   480→        if ($row !== null) {
   481→            $decoded = json_decode($row['hits'], true);
   482→            if (is_array($decoded)) $hits = $decoded;
   483→        }
   484→        $hits = array_values(array_filter(array_map('intval', $hits), static fn ($t) => $now - $t < $windowMs));
   485→        if (count($hits) >= $max) {
   486→            q_exec('UPDATE RateLimit SET hits = ? WHERE key = ?', [json_encode($hits), $key]);
   487→            return false;
   488→        }
   489→        $hits[] = $now;
   490→        if ($row === null) {
   491→            q_exec('INSERT INTO RateLimit (key, hits) VALUES (?,?)', [$key, json_encode($hits)]);
   492→        } else {
   493→            q_exec('UPDATE RateLimit SET hits = ? WHERE key = ?', [json_encode($hits), $key]);
   494→        }
   495→        // Housekeeping ringan.
   496→        if (random_int(1, 100) === 1) {
   497→            q_exec('DELETE FROM RateLimit WHERE key NOT IN (SELECT DISTINCT key FROM RateLimit)', []);
   498→            q_exec("DELETE FROM RateLimit WHERE hits = '[]'", []);
   499→        }
   500→        return true;
   501→    } catch (Throwable $e) {
   502→        return true; // kegagalan limiter tidak boleh memblokir form
   503→    }
   504→}

   506→/* =============================================================== slug/tiket */
   507→function slugify(string $text): string
   508→{
   509→    $t = strtolower(trim($text));
   510→    $t = preg_replace('/[^a-z0-9\s-]/', '', $t) ?? '';
   511→    $t = preg_replace('/[\s_]+/', '-', $t) ?? '';
   512→    $t = preg_replace('/-+/', '-', $t) ?? '';
   513→    return trim($t, '-');
   514→}

   516→const TICKET_CHARS = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
   517→
   518→/** Kode tiket unik MHD-XXXXXX (paritas generateTicketCode Node). */
   519→function generate_ticket_code(): string
   520→{
   521→    for ($attempt = 0; $attempt < 25; $attempt++) {
   522→        $code = 'MHD-';
   523→        for ($i = 0; $i < 6; $i++) {
   524→            $code .= TICKET_CHARS[random_int(0, strlen(TICKET_CHARS) - 1)];
   525→        }
   526→        $exists = q_one('SELECT id FROM MembershipApplication WHERE ticketCode = ?', [$code]);
   527→        if ($exists === null) return $code;
   528→    }
   529→    $n = now_ms();
   530→    $fallback = '';
   531→    for ($i = 0; $i < 6; $i++) {
   532→        $fallback = TICKET_CHARS[$n % strlen(TICKET_CHARS)] . $fallback;
   533→        $n = intdiv($n, strlen(TICKET_CHARS));
   534→    }
   535→    return 'MHD-' . $fallback;
   536→}

   546→/**
   547→ * Terapkan terjemahan konten DB (ContentTranslation) ke baris — paritas
   548→ * applyEntityTranslations Node. Tanpa AI: hanya cache terjemahan yang ada;
   549→ * field tanpa terjemahan tetap memakai teks Indonesia (fallback bawaan).
   550→ */
   551→function apply_translations(array $rows, string $entity, string $locale): array
   552→{
   553→    if ($locale === 'id' || $rows === []) return $rows;
   554→    try {
   555→        $all = q_all('SELECT entityKey, field, value FROM ContentTranslation WHERE entity = ? AND locale = ?', [$entity, $locale]);
   556→    } catch (Throwable $e) {
   557→        return $rows;
   558→    }
   559→    if ($all === []) return $rows;
   560→    $map = [];
   561→    foreach ($all as $t) {
   562→        $map[$t['entityKey']][$t['field']] = $t['value'];
   563→    }
   564→    foreach ($rows as $i => $row) {
   565→        $key = $row['id'] ?? $row['slug'] ?? null;
   566→        if ($key !== null && isset($map[$key]) && is_array($map[$key])) {
   567→            foreach ($map[$key] as $field => $value) {
   568→                if (array_key_exists($field, $row) && $value !== null && $value !== '') {
   569→                    $rows[$i][$field] = $value;
   570→                }
   571→            }
   572→        }
   573→    }
   574→    return $rows;
   575→}

   593→/** Normalisasi nomor Indonesia: 0/8 awal → 62. */
   594→function normalize_wa_number(string $raw): string
   595→{
   596→    $digits = preg_replace('/\D/', '', $raw) ?? '';
   597→    if (str_starts_with($digits, '0')) $digits = '62' . substr($digits, 1);
   598→    elseif (str_starts_with($digits, '8')) $digits = '62' . $digits;
   599→    return $digits;

   602→/** POST ke gateway + penafsiran body (paritas post() Node). Timeout 10 dtk. */
   603→function wa_post(string $url, array $headers, string $bodyRaw): array
   604→{
   605→    $ch = curl_init($url);
   606→    curl_setopt_array($ch, [
   607→        CURLOPT_POST           => true,
   608→        CURLOPT_POSTFIELDS     => $bodyRaw,
   609→        CURLOPT_HTTPHEADER     => $headers,
   610→        CURLOPT_RETURNTRANSFER => true,
   611→        CURLOPT_TIMEOUT        => 10,
   612→        CURLOPT_CONNECTTIMEOUT => 10,
   613→    ]);
   614→    $text = curl_exec($ch);
   615→    $errno = curl_errno($ch);
   616→    $status = (int) curl_getinfo($ch, CURLINFO_HTTP_CODE);
   617→    curl_close($ch);
   618→    if ($errno === CURLE_OPERATION_TIMEDOUT) {
   619→        return ['sent' => false, 'detail' => 'Timeout — gateway tidak merespons 10 detik.'];
   620→    }
   621→    if ($errno !== 0) {
   622→        return ['sent' => false, 'detail' => 'Koneksi gateway gagal (' . $errno . ').'];
   623→    }
   624→    $text = (string) $text;
   625→    if ($status < 200 || $status >= 300) {
   626→        return ['sent' => false, 'detail' => 'HTTP ' . $status . ' — ' . mb_substr($text, 0, 160) ?: 'tanpa isi'];
   627→    }
   628→    $json = json_decode($text, true);
   629→    if (is_array($json) && (isset($json['status']) && ($json['status'] === false || $json['status'] === 'false' || $json['status'] === 0))) {
   630→        return ['sent' => false, 'detail' => (string) ($json['reason'] ?? $json['message'] ?? 'Gateway menolak pesan.')];
   631→    }
   632→    return ['sent' => true, 'detail' => mb_substr($text, 0, 200) ?: 'OK'];
   633→}

   635→/** Kirim pesan WhatsApp sesuai konfigurasi (paritas sendWhatsAppMessage). */
   636→function send_wa_message(string $message): array
   637→{
   638→    try {
   639→        $cfg = get_whatsapp_setting();
   640→        if (!((int) $cfg['enabled'])) return ['sent' => false, 'detail' => 'Notifikasi WhatsApp sedang nonaktif.'];
   641→        if ($cfg['token'] === '') return ['sent' => false, 'detail' => 'Token gateway belum diisi.'];
   642→        if ($cfg['target'] === '') return ['sent' => false, 'detail' => 'Nomor tujuan admin belum diisi.'];
   643→        $target = normalize_wa_number($cfg['target']);
   644→
   645→        switch ($cfg['provider']) {
   646→            case 'FONNTE':
   647→                return wa_post(
   648→                    'https://api.fonnte.com/send',
   649→                    ['Authorization: ' . $cfg['token']],
   650→                    http_build_query(['target' => $target, 'message' => $message])
   651→                );
   652→            case 'WABLAS':
   653→                return wa_post(
   654→                    'https://console.wablas.com/api/send-message?token=' . urlencode($cfg['token']),
   655→                    ['Content-Type: application/json'],
   656→                    json_encode(['phone' => $target, 'message' => $message], JSON_UNESCAPED_UNICODE)
   657→                );
   658→            case 'CUSTOM':
   659→                if ($cfg['apiUrl'] === '') return ['sent' => false, 'detail' => 'URL endpoint CUSTOM belum diisi.'];
   660→                return wa_post(
   661→                    $cfg['apiUrl'],
   662→                    ['Content-Type: application/json'],
   663→                    json_encode(['token' => $cfg['token'], 'target' => $target, 'message' => $message], JSON_UNESCAPED_UNICODE)
   664→                );
   665→            default:
   666→                return ['sent' => false, 'detail' => 'Provider tidak dikenal: ' . $cfg['provider']];
   667→        }
   668→    } catch (Throwable $e) {
   669→        $msg = $e->getMessage();
   670→        return ['sent' => false, 'detail' => str_contains($msg, 'timeout') ? 'Timeout — gateway tidak merespons 10 detik.' : $msg];
   671→    }
   672→}

   724→/** Notifier gagal-aman (paritas notify* Node) — memakai saklar masing-masing. */
   725→function wa_notify(string $kind, array $payload): void
   726→{
   727→    try {
   728→        $cfg = get_whatsapp_setting();
   729→        $enabled = (int) $cfg['enabled'];
   730→        $allowed = $kind === 'application' ? (int) $cfg['notifyApplication'] : (int) $cfg['notifyContact'];
   731→        if (!$enabled || !$allowed) return;
   732→        $text = match ($kind) {
   733→            'contact'     => wa_contact_template($payload),
   734→            'application' => wa_application_template($payload),
   735→            'complaint'   => wa_complaint_template($payload),
   736→            default       => null,
   737→        };
   738→        if ($text === null) return;
   739→        $res = send_wa_message($text);
   740→        q_exec(
   741→            'UPDATE WhatsAppSetting SET lastTestAt = ?, lastTestStatus = ?, updatedAt = ? WHERE id = ?',
   742→            [now_ms(), ($res['sent'] ? 'OK' : 'GAGAL') . ' — ' . mb_substr($res['detail'], 0, 180), now_ms(), $cfg['id']]
   743→        );
   744→    } catch (Throwable $e) {
   745→        // notifikasi tidak boleh mengganggu alur utama
   746→    }
   747→}

   772→function nullable_str(string $key): ?string
   773→{
   774→    $v = body($key);
   775→    if ($v === null) return null;
   776→    if (is_string($v)) return $v === '' ? null : $v;
   777→    return null;

   780→function bool_int(string $key, int $default): int
   781→{
   782→    $v = body($key);
   783→    if (is_bool($v)) return $v ? 1 : 0;
   784→    if (is_numeric($v)) return ((int) $v) ? 1 : 0;
   785→    if (is_string($v)) {
   786→        $lv = strtolower($v);
   787→        if (in_array($lv, ['true', '1', 'yes'], true)) return 1;
   788→        if (in_array($lv, ['false', '0', 'no'], true)) return 0;
   789→    }
   790→    return $default;
   791→}

   793→function int_or(string $key, int $default): int
   794→{
   795→    $v = body($key);
   796→    return is_numeric($v) ? (int) $v : $default;
   797→}
