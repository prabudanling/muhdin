   367→   367→function PermitChecker() {
   368→   368→  const [no, setNo] = useState("");
   369→   369→  const [loading, setLoading] = useState(false);
   370→   370→  const [result, setResult] = useState<VerifyResponse | null>(null);
   371→   371→  const [error, setError] = useState("");
   372→   372→
   373→   373→  const verify = async () => {
   374→   374→    const q = no.trim().toUpperCase();
   375→   375→    if (!q) {
   376→   376→      setResult(null);
   377→   377→      setError("Masukkan nomor izin Nusuk terlebih dahulu.");
   378→   378→      return;
   379→   379→    }
   380→   380→    setLoading(true);
   381→   381→    setError("");
   382→   382→    try {
   383→   383→      const res = await apiGet<VerifyResponse>(`/api/nusuk/verify?no=${encodeURIComponent(q)}`);
   384→   384→      setResult(res);
   385→   385→    } catch (e) {
   386→   386→      setResult(null);
   387→   387→      setError((e as Error).message);
   388→   388→    } finally {
   389→   389→      setLoading(false);
   390→   390→    }
   391→   391→  };
   392→   392→
   393→   393→  const status = result ? PERMIT_STATUS[result.permit.status] || PERMIT_STATUS.EXPIRED : null;
   394→   394→
   395→   395→  return (
   396→   396→    <div className="max-w-3xl mx-auto">
   397→   397→      <Reveal>
   398→   398→        <div className="rounded-3xl border bg-card shadow-sm overflow-hidden">
   399→   399→          <div className="border-b bg-gradient-to-r from-primary/5 via-transparent to-gold/5 p-5 sm:p-6">

   951→   951→
   952→   952→export function NusukView() {
   953→   953→  const [data, setData] = useState<NusukPublicData | null>(null);
   954→   954→  const [error, setError] = useState("");
   955→   955→  const [loading, setLoading] = useState(true);
   956→   956→  const [attempt, setAttempt] = useState(0);
   957→   957→
   958→   958→  useEffect(() => {
   959→   959→    let cancelled = false;
   960→   960→    apiGet<NusukPublicData>("/api/nusuk/public")
   961→   961→      .then((d) => {
   962→   962→        if (!cancelled) setData(d);
   963→   963→      })
   964→   964→      .catch((e) => {
   965→   965→        if (!cancelled) setError((e as Error).message);
   966→   966→      })
   967→   967→      .finally(() => {
   968→   968→        if (!cancelled) setLoading(false);
   969→   969→      });
   970→   970→    return () => {
   971→   971→      cancelled = true;
   972→   972→    };
   973→   973→  }, [attempt]);
   974→   974→
   975→   975→  const retry = () => {
   976→   976→    setError("");
   977→   977→    setLoading(true);
   978→   978→    setAttempt((a) => a + 1);
   979→   979→  };
   980→   980→
   981→   981→  return (
   982→   982→    <div className="flex flex-col">
   983→   983→      <Hero data={data} />
   984→   984→
   985→   985→      {error ? (
   986→   986→        <>
   987→   987→          <ErrorState message={error} onRetry={retry} />
   988→   988→          {/* Checker tetap dapat dipakai karena endpoint-nya independen */}
   989→   989→          <PermitCheckerSection />
   990→   990→          <CtaSection />
   991→   991→        </>
   992→   992→      ) : !data ? (
   993→   993→        <LoadingSkeleton />
   994→   994→      ) : (
   995→   995→        <>
   996→   996→          <MetricsSection metrics={data.metrics} totalSyncs={data.connection.totalSyncs} />
   997→   997→          <PermitCheckerSection />
   998→   998→          <MatrixSection ecosystems={data.ecosystems} />
   999→   999→          <ApiBridgeSection />
  1000→  1000→          <SyncFeedSection logs={data.recentLogs} />
  1001→  1001→          <TopMembersSection members={data.topMembers} />
  1002→  1002→          <CtaSection />
  1003→  1003→        </>
  1004→  1004→      )}
  1005→  1005→    </div>
  1006→  1006→  );
  1007→  1007→}
