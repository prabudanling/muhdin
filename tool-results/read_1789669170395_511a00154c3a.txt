   537→export function HomeView() {
   538→  const [data, setData] = useState<{
   539→    ecosystems: Ecosystem[];
   540→    steps: JourneyStep[];
   541→    roadmap: Roadmap[];
   542→    testimonials: Testimonial[];
   543→    articles: Article[];
   544→  } | null>(null);
   545→  const [error, setError] = useState("");
   546→
   547→  useEffect(() => {
   548→    Promise.all([
   549→      apiGet<Ecosystem[]>("/api/ecosystems"),
   550→      apiGet<JourneyStep[]>("/api/journey"),
   551→      apiGet<Roadmap[]>("/api/roadmap"),
   552→      apiGet<Testimonial[]>("/api/testimonials"),
   553→      apiGet<Article[]>("/api/articles?limit=3"),
   554→    ])
   555→      .then(([ecosystems, steps, roadmap, testimonials, articles]) =>
   556→        setData({ ecosystems, steps, roadmap, testimonials, articles })
   557→      )
   558→      .catch((e) => setError(e.message));
   559→  }, []);
   560→
   561→  if (error) {
   562→    return (
   563→      <div className="py-24 text-center">
   564→        <Icon name="alert-triangle" className="h-10 w-10 mx-auto text-destructive" />
   565→        <p className="mt-3 text-muted-foreground">{error}</p>
   566→        <Button variant="outline" className="mt-4" onClick={() => window.location.reload()}>
   567→          <Icon name="rotate" className="h-4 w-4 mr-2" /> Muat Ulang
   568→        </Button>
   569→      </div>
   570→    );
   571→  }
   572→
   573→  if (!data) return <LoadingSkeleton />;
   574→
   575→  return (
   576→    <div className={cn("flex flex-col")}>
   577→      <Hero />
   578→      <NusukBar />
   579→      <EcosystemSection ecosystems={data.ecosystems} />
   580→      <JourneySection steps={data.steps} />
   581→      <PartnersSection />
   582→      <ValuesSection />
   583→      <TechSection />
   584→      <RoadmapSection roadmap={data.roadmap} />
   585→      <BenefitsSection />
   586→      <TestimonialSection testimonials={data.testimonials} />
   587→      <LatestNews articles={data.articles} />
   588→      <JoinCTA />
   589→    </div>
   590→  );
   591→}
