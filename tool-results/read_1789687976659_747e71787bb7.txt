    27→    27→/* timeAgo versi i18n (label & satuan lewat kamus, angka via formatNumberL10n). */
    28→    28→function timeAgoL10n(
    29→    29→  iso: string | null | undefined,
    30→    30→  t: (key: string, vars?: Record<string, string | number>) => string,
    31→    31→  locale: Locale
    32→    32→) {
    33→    33→  if (!iso) return t("home.nusukLive.belumSinkron");
    34→    34→  try {
    35→    35→    const diff = Date.now() - new Date(iso).getTime();
    36→    36→    const s = Math.max(1, Math.floor(diff / 1000));
    37→    37→    if (s < 60) return t("home.nusukLive.detik", { n: formatNumberL10n(s, locale) });
    38→    38→    const m = Math.floor(s / 60);
    39→    39→    if (m < 60) return t("home.nusukLive.menit", { n: formatNumberL10n(m, locale) });
    40→    40→    const h = Math.floor(m / 60);
    41→    41→    if (h < 24) return t("home.nusukLive.jam", { n: formatNumberL10n(h, locale) });
    42→    42→    const d = Math.floor(h / 24);
    43→    43→    return t("home.nusukLive.hari", { n: formatNumberL10n(d, locale) });
    44→    44→  } catch {
    45→    45→    return iso;
    46→    46→  }
    47→    47→}
