   136→function numL10n(n: number, locale: Locale) {
   137→  return formatNumberL10n(Number(n) || 0, locale);
   138→}

   140→function rateL10n(n: number, locale: Locale) {
   141→  try {
   142→    return new Intl.NumberFormat(locale === "id" ? "id-ID" : locale === "ar" ? "ar" : "en-GB", {
   143→      maximumFractionDigits: 1,
   144→    }).format(Number(n) || 0);
   145→  } catch {
   146→    return String(n);
   147→  }
   148→}

   150→/** Versi l10n dari timeAgo (client-api) — logika sama, teks via kamus. */
   151→function timeAgoL10n(iso: string | null | undefined, t: TFn) {
   152→  if (!iso) return t("nusuk.agoNever");
   153→  try {
   154→    const diff = Date.now() - new Date(iso).getTime();
   155→    const s = Math.max(1, Math.floor(diff / 1000));
   156→    if (s < 60) return t("nusuk.agoSec", { n: s });
   157→    const m = Math.floor(s / 60);
   158→    if (m < 60) return t("nusuk.agoMin", { n: m });
   159→    const h = Math.floor(m / 60);
   160→    if (h < 24) return t("nusuk.agoHour", { n: h });
   161→    const d = Math.floor(h / 24);
   162→    return t("nusuk.agoDay", { n: d });
   163→  } catch {
   164→    return iso ?? "";
   165→  }
   166→}
   167→
   168→function durL10n(ms: number, locale: Locale, t: TFn) {
   169→  if (ms < 1000) return t("nusuk.durMs", { n: numL10n(ms, locale) });
   170→  return t("nusuk.durSec", { n: rateL10n(ms / 1000, locale) });
   171→}

   439→function PermitChecker() {
   440→  const { t, locale } = useT();
   441→  const [no, setNo] = useState("");
   442→  const [loading, setLoading] = useState(false);
   443→  const [result, setResult] = useState<VerifyResponse | null>(null);
   444→  const [error, setError] = useState("");
   445→
   446→  const verify = async () => {
   447→    const q = no.trim().toUpperCase();
   448→    if (!q) {
   449→      setResult(null);
   450→      setError(t("nusuk.errEmptyNo"));
   451→      return;
   452→    }
   453→    setLoading(true);
   454→    setError("");
   455→    try {
   456→      const res = await apiGet<VerifyResponse>(`/api/nusuk/verify?no=${encodeURIComponent(q)}`);
   457→      setResult(res);
   458→    } catch (e) {
   459→      setResult(null);
   460→      setError((e as Error).message);
   461→    } finally {
   462→      setLoading(false);
   463→    }
   464→  };
   465→
   466→  const statusCls = result
   467→    ? PERMIT_STATUS_CLS[result.permit.status] || PERMIT_STATUS_CLS.EXPIRED
   468→    : PERMIT_STATUS_CLS.EXPIRED;
   469→  const statusKey = result
   470→    ? STATUS_KEYS[result.permit.status] || "nusuk.statusEXPIRED"
   471→    : "nusuk.statusEXPIRED";
   472→
   473→  return (
   474→    <div className="max-w-3xl mx-auto">
   475→      <Reveal>

  1045→export function NusukView() {
  1046→  const { locale } = useT();
  1047→  const [data, setData] = useState<NusukPublicData | null>(null);
  1048→  const [error, setError] = useState("");
  1049→  const [loading, setLoading] = useState(true);
  1050→  const [attempt, setAttempt] = useState(0);
  1051→
  1052→  useEffect(() => {
  1053→    let cancelled = false;
  1054→    apiGet<NusukPublicData>(`/api/nusuk/public?locale=${locale}`)
  1055→      .then((d) => {
  1056→        if (!cancelled) setData(d);
  1057→      })
  1058→      .catch((e) => {
  1059→        if (!cancelled) setError((e as Error).message);
  1060→      })
  1061→      .finally(() => {
  1062→        if (!cancelled) setLoading(false);
  1063→      });
  1064→    return () => {
  1065→      cancelled = true;
  1066→    };
  1067→  }, [attempt, locale]);
  1068→
  1069→  const retry = () => {
  1070→    setError("");
  1071→    setLoading(true);
  1072→    setAttempt((a) => a + 1);
  1073→  };
  1074→
  1075→  return (
  1076→    <div className="flex flex-col">
  1077→      <Hero data={data} />
  1078→
  1079→      {error ? (
  1080→        <>
  1081→          <ErrorState message={error} onRetry={retry} />
  1082→          {/* Checker tetap dapat dipakai karena endpoint-nya independen */}
  1083→          <PermitCheckerSection />
  1084→          <CtaSection />
  1085→        </>
  1086→      ) : !data ? (
  1087→        <LoadingSkeleton />
  1088→      ) : (
  1089→        <>
  1090→          <MetricsSection metrics={data.metrics} totalSyncs={data.connection.totalSyncs} />
  1091→          <PermitCheckerSection />
  1092→          <MatrixSection ecosystems={data.ecosystems} />
  1093→          <ApiBridgeSection />
  1094→          <SyncFeedSection logs={data.recentLogs} />
  1095→          <TopMembersSection members={data.topMembers} />
  1096→          <CtaSection />
  1097→        </>
  1098→      )}
  1099→    </div>
  1100→  );
  1101→}
