    36→    36→/* timeAgo versi i18n (label & satuan lewat kamus, angka via formatNumberL10n). */
    37→    37→function timeAgoL10n(
    38→    38→  iso: string | null | undefined,
    39→    39→  t: (key: string, vars?: Record<string, string | number>) => string,
    40→    40→  locale: Locale
    41→    41→) {
    42→    42→  if (!iso) return t("home.nusukLive.belumSinkron");
    43→    43→  try {
    44→    44→    const diff = Date.now() - new Date(iso).getTime();
    45→    45→    const s = Math.max(1, Math.floor(diff / 1000));
    46→    46→    if (s < 60) return t("home.nusukLive.detik", { n: formatNumberL10n(s, locale) });
    47→    47→    const m = Math.floor(s / 60);
    48→    48→    if (m < 60) return t("home.nusukLive.menit", { n: formatNumberL10n(m, locale) });
    49→    49→    const h = Math.floor(m / 60);
    50→    50→    if (h < 24) return t("home.nusukLive.jam", { n: formatNumberL10n(h, locale) });
    51→    51→    const d = Math.floor(h / 24);
    52→    52→    return t("home.nusukLive.hari", { n: formatNumberL10n(d, locale) });
    53→    53→  } catch {
    54→    54→    return iso;
    55→    55→  }
    56→    56→}
