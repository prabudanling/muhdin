    55→/* timeAgo versi i18n (label & satuan lewat kamus, angka via formatNumberL10n). */
    56→function timeAgoL10n(
    57→  iso: string | null | undefined,
    58→  t: (key: string, vars?: Record<string, string | number>) => string,
    59→  locale: Locale
    60→) {
    61→  if (!iso) return t("home.nusukLive.belumSinkron");
    62→  try {
    63→    const diff = Date.now() - new Date(iso).getTime();
    64→    const s = Math.max(1, Math.floor(diff / 1000));
    65→    if (s < 60) return t("home.nusukLive.detik", { n: formatNumberL10n(s, locale) });
    66→    const m = Math.floor(s / 60);
    67→    if (m < 60) return t("home.nusukLive.menit", { n: formatNumberL10n(m, locale) });
    68→    const h = Math.floor(m / 60);
    69→    if (h < 24) return t("home.nusukLive.jam", { n: formatNumberL10n(h, locale) });
    70→    const d = Math.floor(h / 24);
    71→    return t("home.nusukLive.hari", { n: formatNumberL10n(d, locale) });
    72→  } catch {
    73→    return iso;
    74→  }
    75→}
