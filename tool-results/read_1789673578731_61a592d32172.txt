   600→export function HomeView() {
   601→  const [data, setData] = useState<{
   602→    ecosystems: Ecosystem[];
   603→    steps: JourneyStep[];
   604→    roadmap: Roadmap[];
   605→    testimonials: Testimonial[];
   606→    articles: Article[];
   607→  } | null>(null);
   608→  const [error, setError] = useState("");
   609→
   610→  useEffect(() => {
   611→    Promise.all([
   612→      apiGet<Ecosystem[]>("/api/ecosystems"),
   613→      apiGet<JourneyStep[]>("/api/journey"),
   614→      apiGet<Roadmap[]>("/api/roadmap"),
   615→      apiGet<Testimonial[]>("/api/testimonials"),
   616→      apiGet<Article[]>("/api/articles?limit=3"),
   617→    ])
   618→      .then(([ecosystems, steps, roadmap, testimonials, articles]) =>
   619→        setData({ ecosystems, steps, roadmap, testimonials, articles })
   620→      )
   621→      .catch((e) => setError(e.message));
   622→  }, []);
   623→
   624→  if (error) {
   625→    return (
   626→      <div className="py-24 text-center">
   627→        <Icon name="alert-triangle" className="h-10 w-10 mx-auto text-destructive" />
   628→        <p className="mt-3 text-muted-foreground">{error}</p>
   629→        <Button variant="outline" className="mt-4" onClick={() => window.location.reload()}>
   630→          <Icon name="rotate" className="h-4 w-4 mr-2" /> Muat Ulang
   631→        </Button>
   632→      </div>
   633→    );
   634→  }
   635→
   636→  if (!data) return <LoadingSkeleton />;
   637→
   638→  return (
   639→    <div className={cn("flex flex-col")}>
   640→      <Hero />
   641→      <NusukLiveStrip />
   642→      <NusukBar />
   643→      <EcosystemSection ecosystems={data.ecosystems} />
   644→      <JourneySection steps={data.steps} />
   645→      <PartnersSection />
   646→      <ValuesSection />
   647→      <TechSection />
   648→      <RoadmapSection roadmap={data.roadmap} />
   649→      <BenefitsSection />
   650→      <TestimonialSection testimonials={data.testimonials} />
   651→      <LatestNews articles={data.articles} />
   652→      <JoinCTA />
   653→    </div>
   654→  );
   655→}
