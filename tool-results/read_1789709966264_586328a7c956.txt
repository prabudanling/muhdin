   137→function numL10n(n: number, locale: Locale) {
   138→  return formatNumberL10n(Number(n) || 0, locale);
   139→}
   140→
   141→function rateL10n(n: number, locale: Locale) {
   142→  try {
   143→    return new Intl.NumberFormat(locale === "id" ? "id-ID" : locale === "ar" ? "ar" : "en-GB", {
   144→      maximumFractionDigits: 1,
   145→    }).format(Number(n) || 0);
   146→  } catch {
   147→    return String(n);
   148→  }
   149→}
   150→
   151→/** Versi l10n dari timeAgo (client-api) — logika sama, teks via kamus. */
   152→function timeAgoL10n(iso: string | null | undefined, t: TFn) {
   153→  if (!iso) return t("nusuk.agoNever");
   154→  try {
   155→    const diff = Date.now() - new Date(iso).getTime();
   156→    const s = Math.max(1, Math.floor(diff / 1000));
   157→    if (s < 60) return t("nusuk.agoSec", { n: s });
   158→    const m = Math.floor(s / 60);
   159→    if (m < 60) return t("nusuk.agoMin", { n: m });
   160→    const h = Math.floor(m / 60);
   161→    if (h < 24) return t("nusuk.agoHour", { n: h });
   162→    const d = Math.floor(h / 24);
   163→    return t("nusuk.agoDay", { n: d });
   164→  } catch {
   165→    return iso ?? "";
   166→  }
   167→}
   168→
   169→function durL10n(ms: number, locale: Locale, t: TFn) {
   170→  if (ms < 1000) return t("nusuk.durMs", { n: numL10n(ms, locale) });
   171→  return t("nusuk.durSec", { n: rateL10n(ms / 1000, locale) });
   172→}

   440→function PermitChecker() {
   441→  const { t, locale } = useT();
   442→  const [no, setNo] = useState("");
   443→  const [loading, setLoading] = useState(false);
   444→  const [result, setResult] = useState<VerifyResponse | null>(null);
   445→  const [error, setError] = useState("");
   446→
   447→  const verify = async () => {
   448→    const q = no.trim().toUpperCase();
   449→    if (!q) {
   450→      setResult(null);
   451→      setError(t("nusuk.errEmptyNo"));
   452→      return;
   453→    }
   454→    setLoading(true);
   455→    setError("");
   456→    try {
   457→      const res = await apiGet<VerifyResponse>(`/api/nusuk/verify?no=${encodeURIComponent(q)}`);
   458→      setResult(res);
   459→    } catch (e) {
   460→      setResult(null);
   461→      setError((e as Error).message);
   462→    } finally {
   463→      setLoading(false);
   464→    }
   465→  };
   466→
   467→  const statusCls = result
   468→    ? PERMIT_STATUS_CLS[result.permit.status] || PERMIT_STATUS_CLS.EXPIRED
   469→    : PERMIT_STATUS_CLS.EXPIRED;
   470→  const statusKey = result
   471→    ? STATUS_KEYS[result.permit.status] || "nusuk.statusEXPIRED"
   472→    : "nusuk.statusEXPIRED";
   473→

  1046→export function NusukView() {
  1047→  const { locale } = useT();
  1048→  const [data, setData] = useState<NusukPublicData | null>(null);
  1049→  const [error, setError] = useState("");
  1050→  const [loading, setLoading] = useState(true);
  1051→  const [attempt, setAttempt] = useState(0);
  1052→
  1053→  useEffect(() => {
  1054→    let cancelled = false;
  1055→    apiGet<NusukPublicData>(`/api/nusuk/public?locale=${locale}`)
  1056→      .then((d) => {
  1057→        if (!cancelled) setData(d);
  1058→      })
  1059→      .catch((e) => {
  1060→        if (!cancelled) setError((e as Error).message);
  1061→      })
  1062→      .finally(() => {
  1063→        if (!cancelled) setLoading(false);
  1064→      });
  1065→    return () => {
  1066→      cancelled = true;
  1067→    };
  1068→  }, [attempt, locale]);
  1069→
  1070→  const retry = () => {
  1071→    setError("");
  1072→    setLoading(true);
  1073→    setAttempt((a) => a + 1);
  1074→  };
  1075→
  1076→  return (
  1077→    <div className="flex flex-col">
  1078→      <Hero data={data} />
  1079→
  1080→      {error ? (
  1081→        <>
  1082→          <ErrorState message={error} onRetry={retry} />
  1083→          {/* Checker tetap dapat dipakai karena endpoint-nya independen */}
  1084→          <PermitCheckerSection />
  1085→          <CtaSection />
  1086→        </>
  1087→      ) : !data ? (
  1088→        <LoadingSkeleton />
  1089→      ) : (
  1090→        <>
  1091→          <MetricsSection metrics={data.metrics} totalSyncs={data.connection.totalSyncs} />
  1092→          <PermitCheckerSection />
  1093→          <MatrixSection ecosystems={data.ecosystems} />
  1094→          <ApiBridgeSection />
  1095→          <SyncFeedSection logs={data.recentLogs} />
  1096→          <TopMembersSection members={data.topMembers} />
  1097→          <CtaSection />
  1098→        </>
  1099→      )}
  1100→    </div>
  1101→  );
  1102→}
