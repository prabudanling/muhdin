   196→   196→/** Normalisasi satu baris SQLite → bentuk JSON ala Prisma (ISO date, bool, int). */
   197→   197→function cast_row(string $table, array $row): array
   198→   198→{
   199→   199→    foreach (DATE_COLS[$table] ?? [] as $c) {
   200→   200→        if (array_key_exists($c, $row) && $row[$c] !== null) {
   201→   201→            $row[$c] = is_numeric($row[$c]) ? iso_date((int) $row[$c]) : $row[$c];
   202→   202→        }
   203→   203→    }
   204→   204→    foreach (BOOL_COLS[$table] ?? [] as $c) {
   205→   205→        if (array_key_exists($c, $row) && $row[$c] !== null) {
   206→   206→            $row[$c] = (bool) ((int) $row[$c]);
   207→   207→        }
   208→   208→    }
   209→   209→    foreach (INT_COLS[$table] ?? [] as $c) {
   210→   210→        if (array_key_exists($c, $row) && $row[$c] !== null && is_numeric($row[$c])) {
   211→   211→            $row[$c] = (int) $row[$c];
   212→   212→        }
   213→   213→    }
   214→   214→    return $row;
   215→   215→}

   271→   271→/* ================================================================== router */
   272→   272→final class Router
   273→   273→{
   274→   274→    /** @var array<string, array<string, callable>> */
   275→   275→    private array $routes = [];
   276→   276→
   277→   277→    public function on(string $method, string $pattern, callable $handler): void
   278→   278→    {
   279→   279→        $this->routes[strtoupper($method)][$pattern] = $handler;
   280→   280→    }
   281→   281→
   282→   282→    /** Pattern: segmen literal, ':param', dan '*' (sisa segmen). */
   283→   283→    public function dispatch(string $method, string $path): void
   284→   284→    {
   285→   285→        $segs = $path === '' ? [] : explode('/', trim($path, '/'));
   286→   286→        foreach (['ANY', strtoupper($method)] as $m) {
   287→   287→            foreach ($this->routes[$m] ?? [] as $pattern => $handler) {
   288→   288→                $params = $this->match($pattern === '' ? [] : explode('/', $pattern), $segs);
   289→   289→                if ($params !== null) {
   290→   290→                    $handler($params);
   291→   291→                    return;
   292→   292→                }
   293→   293→            }
   294→   294→        }
   295→   295→        fail('Endpoint tidak ditemukan: ' . strtoupper($method) . ' /' . implode('/', $segs), 404);
   296→   296→    }
   297→   297→
   298→   298→    /** @return array<string,string>|null */
   299→   299→    private function match(array $pattern, array $segs): ?array
   300→   300→    {
   301→   301→        $params = [];
   302→   302→        $i = 0;
   303→   303→        foreach ($pattern as $ps) {
   304→   304→            if ($ps === '*') {
   305→   305→                $rest = array_slice($segs, $i);
   306→   306→                if ($rest === []) return null;
   307→   307→                $params['rest'] = implode('/', $rest);
   308→   308→                return $params;
   309→   309→            }
   310→   310→            if (!array_key_exists($i, $segs)) return null;
   311→   311→            if ($ps !== '' && $ps[0] === ':') {
   312→   312→                $params[substr($ps, 1)] = rawurldecode($segs[$i]);
   313→   313→            } elseif ($ps !== $segs[$i]) {
   314→   314→                return null;
   315→   315→            }
   316→   316→            $i++;
   317→   317→        }
   318→   318→        return count($segs) === count($pattern) ? $params : null;
   319→   319→    }
   320→   320→}

   322→   322→/* ================================================================ otentikasi */
   323→   323→/**
   324→   324→ * Mirip getSessionUser(): cookie muhdin_session → baris Session+User.
   325→   325→ * Sesi kedaluwarsa / user nonaktif dibersihkan (paritas Task 15-c).
   326→   326→ * Return baris User penuh (termasuk kolom password — handler yang memilah).
   327→   327→ */
   328→   328→function current_user(): ?array
   329→   329→{
   330→   330→    try {
   331→   331→        $token = read_cookie(MUHDIN_SESSION_COOKIE);
   332→   332→        if ($token === null) return null;
   333→   333→        $row = q_one(
   334→   334→            'SELECT u.*, s.id AS __sessionId, s.expiresAt AS __expiresAt FROM Session s JOIN User u ON u.id = s.userId WHERE s.token = ? LIMIT 1',
   335→   335→            [$token]
   336→   336→        );
   337→   337→        if ($row === null) return null;
   338→   338→        if ((int) $row['__expiresAt'] < now_ms()) {
   339→   339→            q_exec('DELETE FROM Session WHERE id = ?', [$row['__sessionId']]);
   340→   340→            return null;
   341→   341→        }
   342→   342→        if (!((int) $row['isActive'])) {
   343→   343→            q_exec('DELETE FROM Session WHERE userId = ?', [$row['id']]);
   344→   344→            return null;
   345→   345→        }
   346→   346→        return $row;
   347→   347→    } catch (Throwable $e) {
   348→   348→        return null;
   349→   349→    }
   350→   350→}

   352→   352→/**
   353→   353→ * Verifikasi password. Hash aktif di edisi PHP = bcrypt ($2y$, password_hash).
   354→   354→ * Hash scrypt warisan versi Node ("salt:hash") tidak dapat diverifikasi PHP —
   355→   355→ * akun demo di-rehash bcrypt saat build, akun lain direset lewat CMS.
   356→   356→ */
   357→   357→function verify_password(string $password, string $stored): bool
   358→   358→{
   359→   359→    if ($password === '' || $stored === '') return false;
   360→   360→    if (strpos($stored, ':') !== false && strpos($stored, '$') !== 0) {
   361→   361→        return false; // format scrypt Node → tidak didukung PHP
   362→   362→    }
   363→   363→    $hash = str_starts_with($stored, '$2b$') || str_starts_with($stored, '$2a$')
   364→   364→        ? '$2y$' . substr($stored, 4)
   365→   365→        : $stored;
   366→   366→    return password_verify($password, $hash);
   367→   367→}

   369→   369→function hash_password(string $password): string
   370→   370→{
   371→   371→    return password_hash($password, PASSWORD_BCRYPT, ['cost' => 10]);
   372→   372→}

   374→   374→/** Buat sesi baru + set cookie (paritas createSession Node). */
   375→   375→function create_session(string $userId): string
   376→   376→{
   377→   377→    $token = bin2hex(random_bytes(32));
   378→   378→    $now = now_ms();
   379→   379→    q_exec(
   380→   380→        'INSERT INTO Session (id, token, userId, expiresAt, createdAt) VALUES (?,?,?,?,?)',
   381→   381→        [new_id(), $token, $userId, $now + MUHDIN_SESSION_DAYS * 86400000, $now]
   382→   382→    );
   383→   383→    $https = (!empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off')
   384→   384→        || (int) ($_SERVER['SERVER_PORT'] ?? 0) === 443;
   385→   385→    setcookie(MUHDIN_SESSION_COOKIE, $token, [
   386→   386→        'expires'  => time() + MUHDIN_SESSION_DAYS * 86400,
   387→   387→        'path'     => '/',
   388→   388→        'secure'   => $https,
   389→   389→        'httponly' => true,
   390→   390→        'samesite' => 'Lax',
   391→   391→    ]);
   392→   392→    return $token;
   393→   393→}

   472→   472→/** Paritas rateLimit Node: bucket:ip, default 5 hit / 60 detik. */
   473→   473→function rate_limit(string $bucket, int $max = 5, int $windowMs = 60000): bool
   474→   474→{
   475→   475→    try {
   476→   476→        $key = $bucket . ':' . client_ip();
   477→   477→        $now = now_ms();
   478→   478→        $row = q_one('SELECT hits FROM RateLimit WHERE key = ?', [$key]);
   479→   479→        $hits = [];
   480→   480→        if ($row !== null) {
   481→   481→            $decoded = json_decode($row['hits'], true);
   482→   482→            if (is_array($decoded)) $hits = $decoded;
   483→   483→        }
   484→   484→        $hits = array_values(array_filter(array_map('intval', $hits), static fn ($t) => $now - $t < $windowMs));
   485→   485→        if (count($hits) >= $max) {
   486→   486→            q_exec('UPDATE RateLimit SET hits = ? WHERE key = ?', [json_encode($hits), $key]);
   487→   487→            return false;
   488→   488→        }
   489→   489→        $hits[] = $now;
   490→   490→        if ($row === null) {
   491→   491→            q_exec('INSERT INTO RateLimit (key, hits) VALUES (?,?)', [$key, json_encode($hits)]);
   492→   492→        } else {
   493→   493→            q_exec('UPDATE RateLimit SET hits = ? WHERE key = ?', [json_encode($hits), $key]);
   494→   494→        }
   495→   495→        // Housekeeping ringan.
   496→   496→        if (random_int(1, 100) === 1) {
   497→   497→            q_exec('DELETE FROM RateLimit WHERE key NOT IN (SELECT DISTINCT key FROM RateLimit)', []);
   498→   498→            q_exec("DELETE FROM RateLimit WHERE hits = '[]'", []);
   499→   499→        }
   500→   500→        return true;
   501→   501→    } catch (Throwable $e) {
   502→   502→        return true; // kegagalan limiter tidak boleh memblokir form
   503→   503→    }
   504→   504→}

   547→   547→ * Terapkan terjemahan konten DB (ContentTranslation) ke baris — paritas
   548→   548→ * applyEntityTranslations Node. Tanpa AI: hanya cache terjemahan yang ada;
   549→   549→ * field tanpa terjemahan tetap memakai teks Indonesia (fallback bawaan).
   550→   550→ */
   551→   551→function apply_translations(array $rows, string $entity, string $locale): array
   552→   552→{
   553→   553→    if ($locale === 'id' || $rows === []) return $rows;
   554→   554→    try {
   555→   555→        $all = q_all('SELECT entityKey, field, value FROM ContentTranslation WHERE entity = ? AND locale = ?', [$entity, $locale]);
   556→   556→    } catch (Throwable $e) {
   557→   557→        return $rows;
   558→   558→    }
   559→   559→    if ($all === []) return $rows;
   560→   560→    $map = [];
   561→   561→    foreach ($all as $t) {
   562→   562→        $map[$t['entityKey']][$t['field']] = $t['value'];
   563→   563→    }
   564→   564→    foreach ($rows as $i => $row) {
   565→   565→        $key = $row['id'] ?? $row['slug'] ?? null;
   566→   566→        if ($key !== null && isset($map[$key]) && is_array($map[$key])) {
   567→   567→            foreach ($map[$key] as $field => $value) {
   568→   568→                if (array_key_exists($field, $row) && $value !== null && $value !== '') {
   569→   569→                    $rows[$i][$field] = $value;
   570→   570→                }
   571→   571→            }
   572→   572→        }
   573→   573→    }
   574→   574→    return $rows;
   575→   575→}

   602→   602→/** POST ke gateway + penafsiran body (paritas post() Node). Timeout 10 dtk. */
   603→   603→function wa_post(string $url, array $headers, string $bodyRaw): array
   604→   604→{
   605→   605→    $ch = curl_init($url);
   606→   606→    curl_setopt_array($ch, [
   607→   607→        CURLOPT_POST           => true,
   608→   608→        CURLOPT_POSTFIELDS     => $bodyRaw,
   609→   609→        CURLOPT_HTTPHEADER     => $headers,
   610→   610→        CURLOPT_RETURNTRANSFER => true,
   611→   611→        CURLOPT_TIMEOUT        => 10,
   612→   612→        CURLOPT_CONNECTTIMEOUT => 10,
   613→   613→    ]);
   614→   614→    $text = curl_exec($ch);
   615→   615→    $errno = curl_errno($ch);
   616→   616→    $status = (int) curl_getinfo($ch, CURLINFO_HTTP_CODE);
   617→   617→    curl_close($ch);
   618→   618→    if ($errno === CURLE_OPERATION_TIMEDOUT) {
   619→   619→        return ['sent' => false, 'detail' => 'Timeout — gateway tidak merespons 10 detik.'];
   620→   620→    }
   621→   621→    if ($errno !== 0) {
   622→   622→        return ['sent' => false, 'detail' => 'Koneksi gateway gagal (' . $errno . ').'];
   623→   623→    }
   624→   624→    $text = (string) $text;
   625→   625→    if ($status < 200 || $status >= 300) {
   626→   626→        return ['sent' => false, 'detail' => 'HTTP ' . $status . ' — ' . mb_substr($text, 0, 160) ?: 'tanpa isi'];
   627→   627→    }
   628→   628→    $json = json_decode($text, true);
   629→   629→    if (is_array($json) && (isset($json['status']) && ($json['status'] === false || $json['status'] === 'false' || $json['status'] === 0))) {
   630→   630→        return ['sent' => false, 'detail' => (string) ($json['reason'] ?? $json['message'] ?? 'Gateway menolak pesan.')];
   631→   631→    }
   632→   632→    return ['sent' => true, 'detail' => mb_substr($text, 0, 200) ?: 'OK'];
   633→   633→}

   635→   635→/** Kirim pesan WhatsApp sesuai konfigurasi (paritas sendWhatsAppMessage). */
   636→   636→function send_wa_message(string $message): array
   637→   637→{
   638→   638→    try {
   639→   639→        $cfg = get_whatsapp_setting();
   640→   640→        if (!((int) $cfg['enabled'])) return ['sent' => false, 'detail' => 'Notifikasi WhatsApp sedang nonaktif.'];
   641→   641→        if ($cfg['token'] === '') return ['sent' => false, 'detail' => 'Token gateway belum diisi.'];
   642→   642→        if ($cfg['target'] === '') return ['sent' => false, 'detail' => 'Nomor tujuan admin belum diisi.'];
   643→   643→        $target = normalize_wa_number($cfg['target']);
   644→   644→
   645→   645→        switch ($cfg['provider']) {
   646→   646→            case 'FONNTE':
   647→   647→                return wa_post(
   648→   648→                    'https://api.fonnte.com/send',
   649→   649→                    ['Authorization: ' . $cfg['token']],
   650→   650→                    http_build_query(['target' => $target, 'message' => $message])
   651→   651→                );
   652→   652→            case 'WABLAS':
   653→   653→                return wa_post(
   654→   654→                    'https://console.wablas.com/api/send-message?token=' . urlencode($cfg['token']),
   655→   655→                    ['Content-Type: application/json'],
   656→   656→                    json_encode(['phone' => $target, 'message' => $message], JSON_UNESCAPED_UNICODE)
   657→   657→                );
   658→   658→            case 'CUSTOM':
   659→   659→                if ($cfg['apiUrl'] === '') return ['sent' => false, 'detail' => 'URL endpoint CUSTOM belum diisi.'];
   660→   660→                return wa_post(
   661→   661→                    $cfg['apiUrl'],
   662→   662→                    ['Content-Type: application/json'],
   663→   663→                    json_encode(['token' => $cfg['token'], 'target' => $target, 'message' => $message], JSON_UNESCAPED_UNICODE)
   664→   664→                );
   665→   665→            default:
   666→   666→                return ['sent' => false, 'detail' => 'Provider tidak dikenal: ' . $cfg['provider']];
   667→   667→        }
   668→   668→    } catch (Throwable $e) {
   669→   669→        $msg = $e->getMessage();
   670→   670→        return ['sent' => false, 'detail' => str_contains($msg, 'timeout') ? 'Timeout — gateway tidak merespons 10 detik.' : $msg];
   671→   671→    }
   672→   672→}
